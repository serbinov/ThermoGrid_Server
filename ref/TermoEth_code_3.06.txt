// ======================= String Formatting Helper =======================
String Sprintf(const char *format, ...) {
    char loc_buf[256];
    char * temp = loc_buf;
    va_list arg;
    va_list copy;
    va_start(arg, format);
    va_copy(copy, arg);
    int len = vsnprintf(temp, sizeof(loc_buf), format, copy);
    va_end(copy);
    if(len < 0) {
        va_end(arg);
        return "";
    };
    if(len >= sizeof(loc_buf)){
        temp = (char*) malloc(len+1);
        if(temp == NULL) {
            va_end(arg);
            return "";
        }
        len = vsnprintf(temp, len+1, format, arg);
    }
    va_end(arg);
    String ret(temp);
    if(temp != loc_buf){
        free(temp);
    }
    return ret;
}

    for (int i = 0; i < MAX_SENSORS_TOTAL; i++) {

        cfg.narodmonSensorIDs[i][0] = '\0';

}

void applyConfig() {
    Serial.println(F("Applying configuration settings on-the-fly..."));

    static uint8_t lastAppliedWindowSize = 0;

    strncpy(currentPduIPString, currentConfig.pduIPString, sizeof(currentPduIPString) - 1); currentPduIPString[sizeof(currentPduIPString) - 1] = '\0';
    strncpy(currentPduCommand, currentConfig.pduCommand, sizeof(currentPduCommand) - 1); currentPduCommand[sizeof(currentPduCommand) - 1] = '\0';
    alarmTemperatureThreshold_op = currentConfig.alarmTemp;
    for (int i = 0; i < MAX_SENSORS_TOTAL; i++) { tempCalibration[i] = currentConfig.calOffsets[i]; }

    const char* ntp_server_to_use = ntpServerName;
    if (currentConfig.useRemoteServerAsNtp && strlen(currentConfig.remoteServerIP) > 0 && String(currentConfig.remoteServerIP) != "0.0.0.0") {
        ntp_server_to_use = currentConfig.remoteServerIP;
    }
    long gmtOffset_sec = (long)currentConfig.gmtOffsetHours * 3600L;
    configTime(gmtOffset_sec, 0, ntp_server_to_use);
    Serial.printf("Timezone set to GMT%d, NTP server: %s\n", currentConfig.gmtOffsetHours, ntp_server_to_use);

    strncpy(currentRemoteServerIP, currentConfig.remoteServerIP, sizeof(currentRemoteServerIP) - 1); currentRemoteServerIP[sizeof(currentRemoteServerIP) - 1] = '\0';
    currentRemoteSendIntervalSec = currentConfig.remoteSendIntervalSec;
    currentRemoteSendEnabled = currentConfig.remoteSendEnabled;

    currentNarodmonSendEnabled_op = currentConfig.narodmonSendEnabled;
    currentNarodmonSendIntervalSec_op = currentConfig.narodmonSendIntervalSec;
    strncpy(currentNarodmonMac_op, currentConfig.narodmonMac, sizeof(currentNarodmonMac_op) - 1); currentNarodmonMac_op[sizeof(currentNarodmonMac_op) - 1] = '\0';
    for(int i = 0; i < MAX_SENSORS_TOTAL; i++) {
        strncpy(currentNarodmonSensorIDs_op[i], currentConfig.narodmonSensorIDs[i], sizeof(currentNarodmonSensorIDs_op[i]) - 1);
        currentNarodmonSensorIDs_op[i][sizeof(currentNarodmonSensorIDs_op[i]) - 1] = '\0';
    }
    
    if (lastAppliedWindowSize == 0) { // First run
        lastAppliedWindowSize = currentConfig.movingAverageWindowSize;
    } else if (currentConfig.movingAverageWindowSize != lastAppliedWindowSize) {
        Serial.printf("Filter window size changed from %u to %u. Resetting filter history.\n", 
                      lastAppliedWindowSize, currentConfig.movingAverageWindowSize);
        #if USE_MOVING_AVERAGE_FILTER
        for (int i = 0; i < MAX_SENSORS_TOTAL; i++) {
            tempFilterHistoryIndex[i] = 0;
            for (int j = 0; j < MAX_MOVING_AVERAGE_WINDOW_SIZE; j++) {
                tempFilterHistory[i][j] = -999.0f;
            }
        }
        #endif
        lastAppliedWindowSize = currentConfig.movingAverageWindowSize;
    }

    if (currentConfig.graphIntervalSeconds != lastAppliedGraphInterval) {
        Serial.printf("Graph history interval changed from %lu to %lu seconds. Clearing history.\n", lastAppliedGraphInterval, currentConfig.graphIntervalSeconds);
        for (int s = 0; s < MAX_SENSORS_TOTAL; s++) { for (int h = 0; h < HISTORY_SIZE; h++) { historyMaxTemps[s][h] = -200.0f; historyMinTemps[s][h] = 200.0f; }}
        currentHistorySlot = 0; lastHistorySlotUpdateMillis = millis(); overallMax72h = -200.0f; overallMin72h = 200.0f;
    }
    HISTORY_INTERVAL_MS = (unsigned long)currentConfig.graphIntervalSeconds * 1000UL;
    lastAppliedGraphInterval = currentConfig.graphIntervalSeconds;
    Serial.printf("Graph history interval set to %lu ms (%lu seconds)\n", HISTORY_INTERVAL_MS, currentConfig.graphIntervalSeconds);

    forceFullRedraw = true;
    Serial.println(F("Live settings applied."));
}

void loadConfig() {
    EEPROM.begin(EEPROM_SIZE);
    EepromConfig tempCfg;
    EEPROM.get(0, tempCfg);

    if (tempCfg.magic == EEPROM_MAGIC_NUMBER && tempCfg.version == EEPROM_CONFIG_VERSION) {
        Serial.println(F("Valid configuration found in EEPROM."));
        currentConfig = tempCfg;
    } else {
        Serial.printf("No valid config (magic:0x%X ver:%d expected_ver:%d) in EEPROM or version mismatch. Loading defaults.\n", tempCfg.magic, tempCfg.version, EEPROM_CONFIG_VERSION);
        setDefaultConfig(currentConfig);
        Serial.println(F("Saving default config..."));
        EEPROM.put(0, currentConfig);
        if (!EEPROM.commit()) Serial.println(F("EEPROM commit failed after setting defaults!"));
        Serial.println(F("Configuration reset to defaults. Rebooting..."));
        delay(2000);
        ESP.restart();
    }
    EEPROM.end();

    applyConfig();
}

void saveConfig(bool rebootAfter) {
    EEPROM.begin(EEPROM_SIZE);
    currentConfig.magic = EEPROM_MAGIC_NUMBER; currentConfig.version = EEPROM_CONFIG_VERSION;
    EEPROM.put(0, currentConfig);
    bool commit_ok = EEPROM.commit();
    EEPROM.end();

    webServerMessage = commit_ok ? "Settings saved successfully!" : "Settings save FAILED!";

    if (rebootAfter) {
        tft.fillScreen(ST77XX_BLACK); tft.setTextColor(ST77XX_WHITE); tft.setTextSize(2);
        tft.setCursor(20, 140); tft.println(F("Settings Saved!"));
        tft.setCursor(20, 160); tft.println(F("Rebooting..."));
        delay(3000);
        ESP.restart();
    }
}

// ======================= Color Conversion Helper =======================
String rgb565ToHtmlColor(uint16_t color) {
    uint8_t r = (color >> 11) & 0x1F;
    uint8_t g = (color >> 5) & 0x3F;
    uint8_t b = color & 0x1F;
    r = (r * 255) / 31;
    g = (g * 255) / 63;
    b = (b * 255) / 31;
    return Sprintf("#%02X%02X%02X", r, g, b);
}


// ======================= Uptime Formatting =======================
String formatUptime(unsigned long ms) {
    unsigned long seconds = ms / 1000, minutes = seconds / 60, hours = minutes / 60, days = minutes / 1440;
    minutes %= 60; hours %= 24;
    return Sprintf("%luD %02luH:%02luM:%02luS", days, hours, minutes, seconds % 60);
}

// ======================= Time Update =======================
void handleTimeUpdate() {
    if (!eth_connected && !wifiStaIsActive) {
        if (timeIsSynchronized) {
            timeIsSynchronized = false;
            currentTimeString = "NTP No Conn";
            prevTimeString = "";
        }
        return;
    }

    struct tm timeinfo;
    if (getLocalTime(&timeinfo, 10)) { // 10ms timeout to avoid blocking
        if (!timeIsSynchronized) {
            Serial.println("System time synchronized via NTP for the first time.");
            timeIsSynchronized = true;
        }
        char timeStringBuff[10];
        strftime(timeStringBuff, sizeof(timeStringBuff), "%H:%M:%S", &timeinfo);
        currentTimeString = String(timeStringBuff);
    } else {
        if (timeIsSynchronized) {
            // Don't mark as unsynced
        }
        currentTimeString = "Syncing...";
    }
}

    }

    forceFullRedraw = true;

    Serial.printf("--- Setup complete. Final Free Heap: %lu\n", ESP.getFreeHeap());
    delay(2000); 
}


// ======================= Loop =======================
void loop() {
  unsigned long currentMillis = millis();

  if (alarmActive) { digitalWrite(LED_GREEN_GPIO, LOW); digitalWrite(LED_RED_GPIO, HIGH); }
  else {
    digitalWrite(LED_RED_GPIO, LOW);
    bool blink_state = (eth_connected || wifiStaIsActive) ? (currentMillis % 2000) < 1000 : (currentMillis % 334) < 167;
    digitalWrite(LED_GREEN_GPIO, blink_state ? HIGH : LOW);
  }
  
  if (isOtaUpdating) {
    webServer.handleClient();
    delay(10);
    return;
  }
  
  handleNetworkMaintenance();
  webServer.handleClient();
  if (apModeIsActive) dnsServer.processNextRequest();

  handleTimeUpdate();
  handleTemperatureReading();
  updateHistoryData();

  if (currentMillis - lastMaxTempCalcTime >= maxTempCalcInterval || forceFullRedraw) {
    lastMaxTempCalcTime = currentMillis; 
    calculateOverallMinMaxTemperatures();
  }

  handleRemoteDataSend();
  handleNarodmonSend();

  if (currentMillis - lastSerialPrintTime >= serialPrintInterval) {
    lastSerialPrintTime = currentMillis; 
    printTemperatureDataToSerial(); 
  }

  if ((currentMillis - lastDisplayUpdateTime >= displayUpdateInterval) || forceFullRedraw) {
    lastDisplayUpdateTime = currentMillis; 
    updateDynamicInfoOnDisplay(); 
    if (forceFullRedraw) forceFullRedraw = false;
  }

  if (currentMillis - lastSensorDetectTime >= sensorDetectInterval) { 
      lastSensorDetectTime = currentMillis; 
      autoDetectDs18b20(); 
  }
}

// ======================= Network Event Handler =======================
void WiFiEvent(WiFiEvent_t event) {
  switch (event) {
    case ARDUINO_EVENT_ETH_START:
      Serial.println("ETH Started"); networkStatusString = "Eth Starting";
      break;
    case ARDUINO_EVENT_ETH_CONNECTED:
      Serial.println("ETH Connected"); networkStatusString = "Eth Connected";
      break;
    case ARDUINO_EVENT_ETH_GOT_IP:
      Serial.printf("ETH MAC: %s, IPv4: %s\n", ETH.macAddress().c_str(), ETH.localIP().toString().c_str());
      networkStatusString = "Eth: " + ETH.localIP().toString();
      eth_connected = true;
      if (wifiStaIsActive || apModeIsActive) {
          Serial.println("Ethernet connection established. Disabling WiFi module.");
          WiFi.mode(WIFI_OFF);
          wifiStaIsActive = false; apModeIsActive = false;
      }
      forceFullRedraw = true;
      break;
    case ARDUINO_EVENT_ETH_DISCONNECTED:
      Serial.println("ETH Disconnected");
      networkStatusString = "Eth Disconnected";
      eth_connected = false; timeIsSynchronized = false; forceFullRedraw = true;
      break;
    case ARDUINO_EVENT_WIFI_STA_GOT_IP:
        Serial.printf("WiFi STA Got IP: %s, MAC: %s\n", WiFi.localIP().toString().c_str(), WiFi.macAddress().c_str());
        wifiStaIsActive = true;
        networkStatusString = "WiFi: " + WiFi.localIP().toString();
        forceFullRedraw = true;
        break;
    case ARDUINO_EVENT_WIFI_STA_DISCONNECTED:
        Serial.println("WiFi STA Disconnected");
        wifiStaIsActive = false; networkStatusString = "WiFi D/C";
        forceFullRedraw = true;
        break;
    default:
      break;
  }
}

// ======================= Network Initialization & Maintenance =======================
void logAndDisplayNetworkStatus(int y_pos, const char* label, bool success, const char* statusMsg) {
    tft.setCursor(10, y_pos); tft.print(label);
    Serial.printf("Network Init: %s... ", label);
    if (success) { tft.setTextColor(ST77XX_GREEN); tft.print(F("OK.")); Serial.printf("OK. %s\n", statusMsg); }
    else { tft.setTextColor(ST77XX_ORANGE); tft.print(F("Fail/Skip")); Serial.printf("Fail/Skip. %s\n", statusMsg); }
    tft.setTextColor(ST77XX_WHITE);
}

void initializeNetwork() {
    WiFi.onEvent(WiFiEvent);
    
    tft.setCursor(10, 45); tft.print(F("Ethernet Init..."));
    Serial.println(F("Attempting Ethernet connection (priority)..."));
    
    ETH.begin(ETH_PHY_W5500, -1, ETH_CS_GPIO, ETH_INT_GPIO, ETH_RST_GPIO, SPI2_HOST, ETH_SCLK_GPIO, ETH_MISO_GPIO, ETH_MOSI_GPIO);
    if (currentConfig.staticIP != IPAddress(0,0,0,0)) {
        ETH.config(currentConfig.staticIP, currentConfig.gateway, currentConfig.subnet, currentConfig.dnsServer);
    }
    
    unsigned long start_eth_wait = millis();
    while(!eth_connected && (millis() - start_eth_wait < 8000)) delay(100);

    if (eth_connected) {
        logAndDisplayNetworkStatus(45, "Ethernet Init", true, ("Eth: " + ETH.localIP().toString()).c_str());
        Serial.println("Ethernet is primary. WiFi will be disabled.");
        WiFi.mode(WIFI_OFF);
        return;
    } 
    
    logAndDisplayNetworkStatus(45, "Ethernet Init", false, "Failed or unavailable. Trying WiFi...");
    ETH.end(); 

    bool isStaConfigured = currentConfig.wifiEnabled && currentConfig.wifiMode == CFG_WIFI_MODE_STA && strlen(currentConfig.wifiSsid) > 0 && String(currentConfig.wifiSsid) != "YourNetworkSSID";
    if (isStaConfigured) {
        Serial.printf("Attempting WiFi STA connect to: %s\n", currentConfig.wifiSsid);
        WiFi.mode(WIFI_STA);
        if (currentConfig.staticIP != IPAddress(0,0,0,0)) {
             WiFi.config(currentConfig.staticIP, currentConfig.gateway, currentConfig.subnet, currentConfig.dnsServer);
        }
        WiFi.begin(currentConfig.wifiSsid, currentConfig.wifiPassword);
        
        unsigned long wifi_start_time = millis();
        tft.setCursor(10, 60); tft.print(F("WiFi STA Init..."));
        while (WiFi.status() != WL_CONNECTED && (millis() - wifi_start_time < 20000)) {
            delay(500); Serial.print(".");
        }
        Serial.println();

        if (WiFi.status() == WL_CONNECTED) {
            wifiStaIsActive = true;
            logAndDisplayNetworkStatus(60, "WiFi STA Init", true, ("WiFi: " + WiFi.localIP().toString()).c_str());
            return;
        }
    }

    logAndDisplayNetworkStatus(60, "WiFi STA Init", false, "Not configured or failed.");
    Serial.printf("Falling back to WiFi AP Mode: %s\n", currentConfig.wifiApSsid);
    
    apModeIsActive = true;
    WiFi.mode(WIFI_AP);
    IPAddress apIP(192, 168, 4, 1);
    WiFi.softAPConfig(apIP, apIP, IPAddress(255, 255, 255, 0));
    WiFi.softAP(currentConfig.wifiApSsid, strlen(currentConfig.wifiApPassword) > 0 ? currentConfig.wifiApPassword : NULL);
    delay(500);
    
    dnsServer.start(DNS_PORT, "*", apIP);
    networkStatusString = "AP: " + WiFi.softAPIP().toString();
    logAndDisplayNetworkStatus(75, "AP Mode Init", true, networkStatusString.c_str());
}

void handleNetworkMaintenance() {
    if (wifiStaIsActive && WiFi.status() != WL_CONNECTED) {
        Serial.println("WiFi connection lost. Reconnecting...");
        networkStatusString = "WiFi Reconnecting...";
        WiFi.reconnect();
    }
}

  return Sprintf("%02X%02X%02X%02X%02X%02X%02X%02X",
          deviceAddress[0], deviceAddress[1], deviceAddress[2], deviceAddress[3],

}

void autoDetectDs18b20() {
  int oldTotalDetectedSensors = totalDetectedSensors; totalDetectedSensors = 0;
  DeviceAddress tempAddress;

  for (int b = 0; b < NUM_DS18B20_BUSES; b++) {
    oneWire[b].reset_search();
    int sensorsOnThisBus = 0;
    while (sensorsOnThisBus < MAX_SENSORS_PER_BUS && oneWire[b].search(tempAddress)) {
        if (totalDetectedSensors >= MAX_SENSORS_TOTAL) break;
        if (tempAddress[0] == 0x28) { // Only standard 1-wire temp sensors
            memcpy(tempSensorAddresses[totalDetectedSensors], tempAddress, 8);
            sensorsOnThisBus++;
            totalDetectedSensors++;
        }
    }
    detectedSensorCount[b] = sensorsOnThisBus;
  }

  if (oldTotalDetectedSensors != totalDetectedSensors) {
    forceFullRedraw = true;
    for (int i = 0; i < MAX_SENSORS_TOTAL; i++) prevTemperatures[i] = -1000.0;
    for (int i = 0; i < GRAPH_SENSORS_TO_PLOT; i++) prevTemperaturesForGraph[i] = -1000.0;
  }
}

void handleTemperatureReading() {
  unsigned long currentMillis = millis();
  if (!conversionStarted && (currentMillis - lastTempRequestTime >= tempRequestInterval)) {
    if (totalDetectedSensors > 0) {
      for (int b = 0; b < NUM_DS18B20_BUSES; b++) { 
        if (detectedSensorCount[b] > 0) sensors[b].requestTemperatures(); 
      }
      lastTempRequestTime = currentMillis; 
      conversionStarted = true;
    } else {
        for (int i = 0; i < MAX_SENSORS_TOTAL; i++){ 
            if (currentTemperatures[i] != -999.0) currentTemperatures[i] = -999.0; 
        }
        conversionStarted = false;
    }
  }
  else if (conversionStarted && (currentMillis - lastTempRequestTime >= DS18B20_CONVERSION_TIME_MS)) {
    int sensorOverallIndex = 0; 

    for (int b = 0; b < NUM_DS18B20_BUSES; b++) {
      for (int i = 0; i < detectedSensorCount[b]; i++) {
        if (sensorOverallIndex < totalDetectedSensors) {
            float rawTempC = sensors[b].getTempC(tempSensorAddresses[sensorOverallIndex]);

            if (rawTempC == DEVICE_DISCONNECTED_C || rawTempC == -127.0f) {
                for (int j = 0; j < MAX_MOVING_AVERAGE_WINDOW_SIZE; j++) {
                    tempFilterHistory[sensorOverallIndex][j] = -999.0f;
                }
                currentTemperatures[sensorOverallIndex] = -999.0;
            } else {
                #if USE_MOVING_AVERAGE_FILTER
                  uint8_t windowSize = constrain(currentConfig.movingAverageWindowSize, 1, MAX_MOVING_AVERAGE_WINDOW_SIZE);
                  tempFilterHistory[sensorOverallIndex][tempFilterHistoryIndex[sensorOverallIndex]] = rawTempC;
                  tempFilterHistoryIndex[sensorOverallIndex] = (tempFilterHistoryIndex[sensorOverallIndex] + 1) % windowSize;
                  
                  float tempSum = 0;
                  int validReadingsCount = 0;
                  for (int j = 0; j < windowSize; j++) {
                    float historyVal = tempFilterHistory[sensorOverallIndex][j];
                    if (historyVal > -127.0f && historyVal != 85.0f) { // More robust check
                      tempSum += historyVal;
                      validReadingsCount++;
                    }
                  }
                  float finalTemp;
                  if (validReadingsCount > 0) {
                      float averageRawTemp = tempSum / validReadingsCount;
                      finalTemp = roundf((averageRawTemp + tempCalibration[sensorOverallIndex]) * 10.0f) / 10.0f;
                  } else if (rawTempC == 85.0f) {
                      finalTemp = 85.0f;
                  } else {
                      finalTemp = -999.0f;
                  }
                  currentTemperatures[sensorOverallIndex] = finalTemp;
                #else
                  bool isLikely85C_raw = (abs(rawTempC - 85.0f)<0.01f);
                  if (isLikely85C_raw && abs(tempCalibration[sensorOverallIndex]) < 0.01f) {
                    currentTemperatures[sensorOverallIndex] = 85.0f;
                  } else {
                    currentTemperatures[sensorOverallIndex] = roundf((rawTempC + tempCalibration[sensorOverallIndex]) * 10.0f) / 10.0f;
                  }
                #endif
            }
        }
        sensorOverallIndex++;
      }
    }
    
    // Alarm logic with Hysteresis
    bool should_be_alarming = false;
    for (int i = 0; i < totalDetectedSensors; i++) {
        float temp = currentTemperatures[i];
        if (temp > -127.0f && temp != 85.0f && temp >= alarmTemperatureThreshold_op) {
            should_be_alarming = true;
            break;
        }
    }

    if (should_be_alarming) {
        alarmActive = true;
    } else if (alarmActive) {
        bool can_deactivate = true;
        for (int i = 0; i < totalDetectedSensors; i++) {
            float temp = currentTemperatures[i];
            if (temp > -127.0f && temp != 85.0f && temp >= (alarmTemperatureThreshold_op - ALARM_HYSTERESIS_C)) {
                can_deactivate = false;
                break;
            }
        }
        if (can_deactivate) alarmActive = false;
    }
    
    if (alarmActive && !pduCommandSent) { sendPduCommand(); pduCommandSent = true; }
    else if (!alarmActive && pduCommandSent) { pduCommandSent = false; }

    for (int i = totalDetectedSensors; i < MAX_SENSORS_TOTAL; i++) { 
      if (currentTemperatures[i] != -999.0) currentTemperatures[i] = -999.0; 
    }
    conversionStarted = false;
  }
}

void printTemperatureDataToSerial() {
    Serial.printf("T:%lu | Up:%s | Sens:%d | NTP: %s | Net: %s | Alarm:%s | Min/Max Hist: ",
        millis(), formatUptime(millis()).c_str(), totalDetectedSensors,
        timeIsSynchronized ? currentTimeString.c_str() : "Sync...",
        networkStatusString.c_str(), alarmActive ? "ON" : "OFF" );
    Serial.printf(overallMin72h < 199.0f ? "%.1f/" : "N/A/", overallMin72h);
    Serial.printf(overallMax72h > -199.0f ? "%.1fC | " : "N/AC | ", overallMax72h);
    Serial.printf("Heap: %.1fKB | ", (float)ESP.getFreeHeap()/1024.0f);
    if (totalDetectedSensors == 0) Serial.println(F("No sensors."));
    else {
        for (int i = 0; i < totalDetectedSensors; i++) {
            Serial.printf("S%d", i + 1);
            if (currentTemperatures[i] <= -127.0) Serial.print(":N/A");
            else if (currentTemperatures[i] == 85.0f && abs(tempCalibration[i])<0.01f) Serial.print(":85.0?");
            else Serial.printf(":%.1fC", currentTemperatures[i]);
            if (i < totalDetectedSensors - 1) Serial.print(F(" | "));
        }
        Serial.println();
    }
}

void sendPduCommand() {
    Serial.printf("ALARM! Sending command '%s' to PDU at %s\n", currentPduCommand, currentPduIPString);
    IPAddress pduIP;
    if (!pduIP.fromString(currentPduIPString) || pduIP == IPAddress(0,0,0,0) || strlen(currentPduCommand) == 0) {
        Serial.println(F("PDU command skipped: Invalid IP or empty command.")); 
        return; 
    }
    if (!eth_connected && !wifiStaIsActive) {
        Serial.println(F("PDU command skipped: No network connection.")); 
        return;
    }

    WiFiClient pduClient;
    if (pduClient.connect(pduIP, 80)) {
        pduClient.printf("GET /?%s HTTP/1.1\r\nHost: %s\r\nConnection: close\r\n\r\n", currentPduCommand, currentPduIPString);
        pduClient.stop();
    } else {
        Serial.println(F("PDU command failed: Could not connect to PDU."));
    }
}

void handleRemoteDataSend() {
    if (!currentRemoteSendEnabled) {

        lastRemoteSendStatus = RSS_DISABLED;
        return;
    }
    
    if (currentRemoteSendIntervalSec == 0 || millis() - lastRemoteSendTime < (currentRemoteSendIntervalSec * 1000UL)) return;
    
    lastRemoteSendTime = millis();
    lastRemoteSendStatus = RSS_PENDING;


    IPAddress serverIP;
    if (!serverIP.fromString(currentRemoteServerIP) || serverIP == IPAddress(0,0,0,0) || (!eth_connected && !wifiStaIsActive)) {
        if (currentRemoteSendEnabled) Serial.println("Remote send skipped: No network or invalid server IP.");
        lastRemoteSendStatus = RSS_FAIL;
        return;
    }

    WiFiClient dataClient;
  uint16_t serverPort = currentConfig.remoteServerPort;

    if (dataClient.connect(serverIP, serverPort)) {
        char payload_buffer[2048]; 
        int offset = 0;
        
        const char* network_type_str = eth_connected ? "ethernet" : (wifiStaIsActive ? "wifi" : "unknown");
        String ip_address_str = eth_connected ? ETH.localIP().toString() : (wifiStaIsActive ? WiFi.localIP().toString() : "N/A");
        char mac_str[18];
        sprintf(mac_str, "%02X:%02X:%02X:%02X:%02X:%02X", device_mac[0], device_mac[1], device_mac[2], device_mac[3], device_mac[4], device_mac[5]);

        time_t now_utc = timeIsSynchronized ? time(nullptr) : 0;
        
        String safeDeviceName = String(currentConfig.deviceName);
        safeDeviceName.replace("\"", "\\\""); safeDeviceName.replace("\\", "\\\\");
        char alarm_thresh_str[8]; dtostrf(currentConfig.alarmTemp, 4, 1, alarm_thresh_str);
        
        offset = snprintf(payload_buffer, sizeof(payload_buffer), "{"
            "\"protocol_version\":\"2.1\",\"deviceId\":\"%s\",\"deviceName\":\"%s\",\"firmware_version\":\"%s\","
            "\"timestamp_utc\":%lu,\"uptime_ms\":%lu,\"geo\":{\"lat\":%.4f,\"lon\":%.4f},"
            "\"status\":{\"alarm_active\":%s,\"alarm_threshold_c\":%s,\"network\":\"%s\",\"ip_address\":\"%s\"},"
            "\"sensors\":[",
            mac_str, safeDeviceName.c_str(), FW_VERSION,
            (unsigned long)now_utc, millis(), currentConfig.geoLat, currentConfig.geoLon,
            alarmActive ? "true" : "false", alarm_thresh_str, network_type_str, ip_address_str.c_str() );

        bool firstSensor = true;
        for (int i = 0; i < totalDetectedSensors; i++) {
            if (currentTemperatures[i] > -127.0f && currentTemperatures[i] != 85.0f) {
                if (!firstSensor) offset += snprintf(payload_buffer + offset, sizeof(payload_buffer) - offset, ",");
                
                char temp_str[8]; dtostrf(currentTemperatures[i], 4, 1, temp_str);
                String safeSensorName = String(currentConfig.sensorNames[i]);
                safeSensorName.replace("\"", "\\\""); safeSensorName.replace("\\", "\\\\");
                if (safeSensorName.length() == 0) safeSensorName = Sprintf("Sensor %d", i);

                offset += snprintf(payload_buffer + offset, sizeof(payload_buffer) - offset, 
                    "{\"id\":%d,\"name\":\"%s\",\"data\":{\"temperature_c\":%s}}", i, safeSensorName.c_str(), temp_str);
                firstSensor = false;
            }
        }
        offset += snprintf(payload_buffer + offset, sizeof(payload_buffer) - offset, "]}");
        
        dataClient.printf("POST / HTTP/1.1\r\nHost: %s\r\nUser-Agent: %s/%s\r\nContent-Type: application/json\r\nContent-Length: %d\r\nConnection: close\r\n\r\n", 
            currentRemoteServerIP, currentConfig.deviceName, FW_VERSION, (int)strlen(payload_buffer));
        dataClient.print(payload_buffer);
        dataClient.stop();
        lastRemoteSendStatus = RSS_OK;
        Serial.printf("Sent remote data packet (v2.1 format) to %s:%u.\n", serverIP.toString().c_str(), serverPort);
    } else {
        lastRemoteSendStatus = RSS_FAIL;
        Serial.printf("Remote data send failed: Could not connect to server at %s:%u.\n", serverIP.toString().c_str(), serverPort);
    }
}

void handleNarodmonSend() {
    if (!currentNarodmonSendEnabled_op || currentNarodmonSendIntervalSec_op < 240 || millis() - lastNarodmonSendTime < (currentNarodmonSendIntervalSec_op * 1000UL)) return;
    lastNarodmonSendTime = millis();

    if (!eth_connected && !wifiStaIsActive) {
        Serial.println(F("Narodmon: Send skipped, no network.")); return;
    }
    char macFormatted[13] = {0}; int j = 0;
    for (int i = 0; currentNarodmonMac_op[i] != '\0' && j < 12; i++) { if (isxdigit(currentNarodmonMac_op[i])) { if (currentNarodmonMac_op[i] != ':') macFormatted[j++] = toupper(currentNarodmonMac_op[i]); } }
    if (j != 12) { Serial.println(F("Narodmon: MAC address invalid.")); return; }
    
    WiFiClient narodmonClient;
    const char* narodmonHost = "narodmon.ru";

    if (narodmonClient.connect(narodmonHost, 8283)) {
        Serial.printf("Narodmon: Connected to %s. Sending data.\n", narodmonHost);
        narodmonClient.printf("#%s", macFormatted); int sensorsInPayload = 0;
        for (int i = 0; i < totalDetectedSensors; i++) {
            if (strlen(currentNarodmonSensorIDs_op[i]) > 0) {
                float temp = currentTemperatures[i];
                if (temp > -127.0f && temp != 85.0f) {
                    char tempStr[8]; dtostrf(temp, 4, 1, tempStr);
                    narodmonClient.printf("#%s#%s", currentNarodmonSensorIDs_op[i], tempStr);
                    if (strlen(currentConfig.sensorNames[i]) > 0) narodmonClient.printf("#%s", currentConfig.sensorNames[i]);
                    sensorsInPayload++;
                }
            }
        }
        if (sensorsInPayload > 0) narodmonClient.println("#DONE");
        narodmonClient.stop();
    } else {
        Serial.printf("Narodmon: Failed to connect to %s.\n", narodmonHost);
    }
}

    for (int i = 0; i < min(totalDetectedSensors, MAX_DISPLAY_SENSOR_BOXES); i++) {
        if (abs(currentTemperatures[i] - prevTemperatures[i]) > 0.05 || (currentTemperatures[i] <= -127.0f) != (prevTemperatures[i] <= -127.0f) || forceFullRedraw) {
            tft.fillRect(current_val_box_x + 1, SENSOR_BOX_Y_START + SENSOR_BOX_VALUE_Y_OFFSET - 2, SENSOR_BOX_WIDTH - 2, SENSOR_BOX_HEIGHT - (SENSOR_BOX_VALUE_Y_OFFSET-2) -1, ST77XX_BLACK);
            char tempDispStr[10]; uint16_t tempColor;
            if (currentTemperatures[i] <= -127.0f) { strcpy(tempDispStr, "N/A"); tempColor = ST77XX_RED; }
            else if (currentTemperatures[i] == 85.0f) { strcpy(tempDispStr, "85.0?"); tempColor = ST77XX_ORANGE; }
            else {
                dtostrf(currentTemperatures[i], 3, 1, tempDispStr);
                if (currentTemperatures[i] >= alarmTemperatureThreshold_op) tempColor = ST77XX_MAGENTA;
                else if (currentTemperatures[i] >= currentConfig.graphTempThresholds[2]) tempColor = currentConfig.graphTempColors[3];
                else if (currentTemperatures[i] >= currentConfig.graphTempThresholds[1]) tempColor = currentConfig.graphTempColors[2];
                else if (currentTemperatures[i] >= currentConfig.graphTempThresholds[0]) tempColor = currentConfig.graphTempColors[1];
                else tempColor = currentConfig.graphTempColors[0];
            }
            tft.setTextSize(2); tft.setTextColor(tempColor);
            int16_t x1, y1; uint16_t w, h; tft.getTextBounds(tempDispStr, 0, 0, &x1, &y1, &w, &h);
            tft.setCursor(current_val_box_x + (SENSOR_BOX_WIDTH - w)/2, SENSOR_BOX_Y_START + SENSOR_BOX_VALUE_Y_OFFSET); tft.print(tempDispStr);
            prevTemperatures[i] = currentTemperatures[i];
        }
        current_val_box_x += SENSOR_BOX_WIDTH + SENSOR_BOX_SPACING;
    }
  }
  
  updateTftField(prevTimeString, currentTimeString, INFO_VALUE_X_OFFSET, INFO_LINES_Y_START, TFT_WIDTH_ST7789 - INFO_VALUE_X_OFFSET, INFO_LINE_HEIGHT - 1, ST77XX_WHITE, ST77XX_BLACK, 1);
  updateTftField(prevNetworkStatusDisplay, networkStatusString, INFO_VALUE_X_OFFSET, INFO_LINES_Y_START + INFO_LINE_HEIGHT, TFT_WIDTH_ST7789 - INFO_VALUE_X_OFFSET, INFO_LINE_HEIGHT - 1, (eth_connected || wifiStaIsActive) ? ST77XX_GREEN : ST77XX_ORANGE, ST77XX_BLACK, 1);
  char alarmTempBuf[8]; dtostrf(alarmTemperatureThreshold_op, 3, 1, alarmTempBuf); strcat(alarmTempBuf, "C");
  updateTftField(prevAlarmTempString, String(alarmTempBuf), INFO_VALUE_X_OFFSET, INFO_LINES_Y_START + 2 * INFO_LINE_HEIGHT, TFT_WIDTH_ST7789 - INFO_VALUE_X_OFFSET, INFO_LINE_HEIGHT - 1, ST77XX_WHITE, ST77XX_BLACK, 1);
  updateTftField(prevUptimeString, formatUptime(millis()), INFO_VALUE_X_OFFSET, INFO_LINES_Y_START + 3 * INFO_LINE_HEIGHT, TFT_WIDTH_ST7789 - INFO_VALUE_X_OFFSET, INFO_LINE_HEIGHT - 1, ST77XX_WHITE, ST77XX_BLACK, 1);

  String remoteInfoStr, prevRemoteSendInfoString; 
  if (lastRemoteSendStatus == RSS_DISABLED) {
      remoteInfoStr = "Disabled";
  } else {
      long countdown_s = -1;
      unsigned long interval_ms = (unsigned long)currentConfig.remoteSendIntervalSec * 1000UL;
      if (interval_ms > 0 && millis() >= lastRemoteSendTime) {
          unsigned long elapsed_ms = millis() - lastRemoteSendTime;
          if (elapsed_ms < interval_ms) countdown_s = (interval_ms - elapsed_ms) / 1000;
      }
      String ip_display = (String(currentConfig.remoteServerIP) == "0.0.0.0") ? "No IP" : String(currentConfig.remoteServerIP);
      uint16_t status_color = ST77XX_WHITE; const char* status_str = "N/A";
      switch(lastRemoteSendStatus) {
          case RSS_OK: status_str = "OK"; status_color = ST77XX_GREEN; break;
          case RSS_FAIL: status_str = "Fail"; status_color = ST77XX_RED; break;
          case RSS_PENDING: status_str = "Sending"; status_color = ST77XX_YELLOW; break;
          default: break;
      }
      String part1 = (countdown_s >= 0) ? Sprintf("%s|next:%lds", ip_display.c_str(), countdown_s) : Sprintf("%s|due", ip_display.c_str());
      String part2 = Sprintf("|%s", status_str);
      tft.fillRect(INFO_VALUE_X_OFFSET - 20, REMOTE_INFO_LINE_Y, TFT_WIDTH_ST7789 - (INFO_VALUE_X_OFFSET-20), INFO_LINE_HEIGHT - 1, ST77XX_BLACK);
      tft.setTextColor(ST77XX_WHITE, ST77XX_BLACK); tft.setTextSize(1);
      tft.setCursor(INFO_VALUE_X_OFFSET - 20, REMOTE_INFO_LINE_Y); tft.print(part1);
      tft.setTextColor(status_color, ST77XX_BLACK); tft.print(part2);
  }
  
  bool graphDataChanged = false;
  for (int i=0; i < min((int)GRAPH_SENSORS_TO_PLOT, totalDetectedSensors); i++) {
    if (abs(currentTemperatures[i] - prevTemperaturesForGraph[i]) > 0.05 || (currentTemperatures[i] <= -127.0f) != (prevTemperaturesForGraph[i] <= -127.0f)) {
        graphDataChanged = true; break;
    }
  }

  float potentialGraphMin = 1000.0, potentialGraphMax = -1000.0; int validTempsForScale = 0;
  for (int i = 0; i < min((int)GRAPH_SENSORS_TO_PLOT, totalDetectedSensors); i++) {
    if (currentTemperatures[i] > -127.0 && currentTemperatures[i] != 85.0f) {
      if (currentTemperatures[i] < potentialGraphMin) potentialGraphMin = currentTemperatures[i];
      if (currentTemperatures[i] > potentialGraphMax) potentialGraphMax = currentTemperatures[i]; validTempsForScale++;
    }
  }
  float newGraphDisplayMin, newGraphDisplayMax;
  if(validTempsForScale > 0) {
    newGraphDisplayMin = floor(potentialGraphMin / 5.0) * 5.0 - 5;
    if (newGraphDisplayMin < 0 && potentialGraphMin >=0) newGraphDisplayMin = 0;
    newGraphDisplayMin = max(newGraphDisplayMin, -50.0f);
    newGraphDisplayMax = ceil(potentialGraphMax / 5.0) * 5.0 + 5;
    newGraphDisplayMax = min(newGraphDisplayMax, 100.0f);
    if (newGraphDisplayMax - newGraphDisplayMin < 10) newGraphDisplayMax = newGraphDisplayMin + 10;
  } else { newGraphDisplayMin = 0.0; newGraphDisplayMax = 50.0; }

  bool graphScaleValuesChanged = (abs(newGraphDisplayMin - actualGraphScaleMin) > 0.1 || abs(newGraphDisplayMax - actualGraphScaleMax) > 0.1);

  if (graphDataChanged || forceFullRedraw || graphScaleValuesChanged) {
    int graphContentY = GRAPH_AREA_Y_START + 1;
    int graphBorderActualHeight = (TFT_HEIGHT_ST7789 - (INFO_LINE_HEIGHT + 2)) - GRAPH_AREA_Y_START;
    drawTemperatureBarGraph(GRAPH_BORDER_X + 1, graphContentY, GRAPH_BORDER_W - 2, graphBorderActualHeight - 2, newGraphDisplayMin, newGraphDisplayMax);
    actualGraphScaleMin = newGraphDisplayMin; actualGraphScaleMax = newGraphDisplayMax;
    for (int i=0; i < GRAPH_SENSORS_TO_PLOT; i++) prevTemperaturesForGraph[i] = currentTemperatures[i];
  }
  
  long currentHeap = ESP.getFreeHeap();
  if (abs(currentHeap - prevHeap) > 100 || forceFullRedraw) {
    char heapBuf[15];
    sprintf(heapBuf, "%.1f KB", (float)currentHeap / 1024.0f);
    tft.fillRect(INFO_VALUE_X_OFFSET - 30, TFT_HEIGHT_ST7789 - 8, TFT_WIDTH_ST7789 - (INFO_VALUE_X_OFFSET - 30), INFO_LINE_HEIGHT, ST77XX_BLACK);

    tft.setTextColor(ST77XX_WHITE, ST77XX_BLACK);
    tft.setTextSize(1);
    tft.setCursor(INFO_VALUE_X_OFFSET - 30, TFT_HEIGHT_ST7789 - 8);
    tft.print(heapBuf);
    prevHeap = currentHeap;
  }
}

void drawTemperatureBarGraph(int x, int y, int w, int h, float displayMinTemp, float displayMaxTemp) {
    tft.fillRect(x, y, w, h, ST77XX_BLACK);
    int16_t x_bounds, y_bounds; uint16_t w_bounds, h_bounds;
    int numSensorsToDisplay = min((int)GRAPH_SENSORS_TO_PLOT, totalDetectedSensors);
    if (numSensorsToDisplay == 0) {
        tft.setTextColor(ST77XX_WHITE); tft.setTextSize(1);
        tft.getTextBounds("No data for graph.", 0, 0, &x_bounds, &y_bounds, &w_bounds, &h_bounds);
        tft.setCursor(x + (w - w_bounds)/2, y + (h - h_bounds) / 2); tft.print(F("No data for graph."));
        return;
    }
    float tempRange = (displayMaxTemp - displayMinTemp < 1.0f) ? 1.0f : (displayMaxTemp - displayMinTemp);
    int barSpacing = 4;
    int totalBarAndSpaceWidth = w / numSensorsToDisplay;
    int barWidth = max(5, totalBarAndSpaceWidth - barSpacing);
    if (barWidth < 5) barSpacing = (w - (barWidth * numSensorsToDisplay)) / (numSensorsToDisplay +1);
    barSpacing = max(1, barSpacing);

    int totalUsedWidth = (barWidth * numSensorsToDisplay) + (barSpacing * (numSensorsToDisplay -1));
    int currentBarX_start = x + (w - totalUsedWidth) / 2;
    tft.setTextSize(1);
    tft.getTextBounds("S1", 0, 0, &x_bounds, &y_bounds, &w_bounds, &h_bounds);
    uint16_t sensorLabelHeight = h_bounds + 2;
    int graphEffectiveHeight = (h > sensorLabelHeight + 10) ? h - sensorLabelHeight : h;

    if (alarmTemperatureThreshold_op >= displayMinTemp && alarmTemperatureThreshold_op <= displayMaxTemp) {
        int alarmLineY = y + graphEffectiveHeight - (int)(((alarmTemperatureThreshold_op - displayMinTemp) / tempRange) * graphEffectiveHeight);
        alarmLineY = constrain(alarmLineY, y, y + graphEffectiveHeight - 1);
        for (int i_line = x; i_line < x + w; i_line += 4) tft.drawFastHLine(i_line, alarmLineY, 2, ST77XX_MAGENTA);
    }

    for (int i = 0; i < numSensorsToDisplay; i++) {
        if (i >= MAX_SENSORS_TOTAL) break;
        int currentBarX = currentBarX_start + i * (barWidth + barSpacing);
        float temp = currentTemperatures[i];
        
        if (sensorLabelHeight > 0) {
            char sensorLabel[4]; sprintf(sensorLabel, "S%d", i + 1);
            tft.setTextSize(1); tft.setTextColor(ST77XX_WHITE, ST77XX_BLACK);
            tft.getTextBounds(sensorLabel, 0, 0, &x_bounds, &y_bounds, &w_bounds, &h_bounds);
            tft.setCursor(currentBarX + (barWidth / 2) - (w_bounds / 2), y + graphEffectiveHeight + 1); tft.print(sensorLabel); 
        }
        if (temp <= -127.0f || temp == 85.0f) {
            tft.setTextSize(1); tft.setTextColor(ST77XX_RED);
            const char* val_str = (temp == 85.0f) ? "85?C" : "N/A";
            tft.getTextBounds(val_str, 0, 0, &x_bounds, &y_bounds, &w_bounds, &h_bounds);
            tft.setCursor(currentBarX + (barWidth / 2) - (w_bounds / 2), y + graphEffectiveHeight - h_bounds - 2); tft.print(val_str);
            continue;
        }
        uint16_t barColor;
        if (temp < currentConfig.graphTempThresholds[0]) barColor = currentConfig.graphTempColors[0];
        else if (temp < currentConfig.graphTempThresholds[1]) barColor = currentConfig.graphTempColors[1];
        else if (temp < currentConfig.graphTempThresholds[2]) barColor = currentConfig.graphTempColors[2];
        else barColor = currentConfig.graphTempColors[3];
        int barPixelHeight = constrain((int)(((temp - displayMinTemp) / tempRange) * graphEffectiveHeight), 0, graphEffectiveHeight);
        int barRectY = y + graphEffectiveHeight - barPixelHeight;
        tft.fillRect(currentBarX, barRectY, barWidth, barPixelHeight, barColor);

        char tempStr[8]; dtostrf(temp, 3, 1, tempStr);
        tft.setTextSize(1); tft.getTextBounds(tempStr, 0, 0, &x_bounds, &y_bounds, &w_bounds, &h_bounds);
        int textX = currentBarX + (barWidth / 2) - (w_bounds / 2);
        int textY = barRectY - h_bounds - 2;
        uint16_t textColor = ST77XX_WHITE, textBgColor = ST77XX_BLACK;
        if (barPixelHeight > h_bounds + 4 && (barRectY + 2 + h_bounds < y + graphEffectiveHeight)) {
            textY = barRectY + 2; textBgColor = barColor;
            int r = (barColor >> 11) & 0x1F, g = (barColor >> 5) & 0x3F, b = barColor & 0x1F;
            if ((r * 299 + g * 587 + b * 114) / 1000 > 128) { textColor = ST77XX_BLACK; } else { textColor = ST77XX_WHITE; }
        }
        if (textY < y) textY = y + 2;
        tft.setTextColor(textColor, textBgColor); tft.setCursor(textX, textY); tft.print(tempStr);
    }

    tft.setTextSize(1); tft.setTextColor(ST77XX_WHITE, ST77XX_BLACK);
    char scale_str[10];
    if (graphEffectiveHeight > 20) {
      dtostrf(displayMaxTemp,3,0,scale_str); strcat(scale_str, "C"); tft.setCursor(x + 2, y + 2); tft.print(scale_str);
      dtostrf(displayMinTemp,3,0,scale_str); strcat(scale_str, "C"); tft.getTextBounds(scale_str, 0, 0, &x_bounds, &y_bounds, &w_bounds, &h_bounds); tft.setCursor(x + 2, y + graphEffectiveHeight - h_bounds - 1); tft.print(scale_str);
    }
}
void updateHistoryData() {
    unsigned long currentMillis_local = millis();
    if (HISTORY_INTERVAL_MS == 0) return;
    if (currentMillis_local - lastHistorySlotUpdateMillis >= HISTORY_INTERVAL_MS) {
        int slots_to_advance = (currentMillis_local - lastHistorySlotUpdateMillis) / HISTORY_INTERVAL_MS;

        slots_to_advance = min(slots_to_advance, HISTORY_SIZE); // Cap advancement

        for (int p=0; p < slots_to_advance; p++) {
            currentHistorySlot = (currentHistorySlot + 1) % HISTORY_SIZE;
            for (int s = 0; s < MAX_SENSORS_TOTAL; s++) { historyMaxTemps[s][currentHistorySlot] = -200.0f; historyMinTemps[s][currentHistorySlot] = 200.0f; }
        }
        lastHistorySlotUpdateMillis += (unsigned long)slots_to_advance * HISTORY_INTERVAL_MS;
    }
    for (int i = 0; i < totalDetectedSensors; i++) {
        float temp = currentTemperatures[i];
        if (temp > -127.0f && temp != 85.0f) {
            if (temp > historyMaxTemps[i][currentHistorySlot]) historyMaxTemps[i][currentHistorySlot] = temp;
            if (temp < historyMinTemps[i][currentHistorySlot]) historyMinTemps[i][currentHistorySlot] = temp;
        }
    }
}
void calculateOverallMinMaxTemperatures() {
    overallMax72h = -200.0f; overallMin72h = 200.0f;

    if (HISTORY_INTERVAL_MS == 0) return;
    int valid_intervals_since_boot = (int)(millis() / HISTORY_INTERVAL_MS);
    int current_slots_filled = min(valid_intervals_since_boot + 1, HISTORY_SIZE);

    for (int h_offset = 0; h_offset < current_slots_filled; h_offset++) {
        int slotIndex = (currentHistorySlot - h_offset + HISTORY_SIZE) % HISTORY_SIZE;
        for (int s = 0; s < totalDetectedSensors; s++) {
            if (historyMaxTemps[s][slotIndex] > overallMax72h) overallMax72h = historyMaxTemps[s][slotIndex];
            if (historyMinTemps[s][slotIndex] < overallMin72h) overallMin72h = historyMinTemps[s][slotIndex];
        }
    }
}

void handleDataJson() {
    char json_buffer[1024]; // Use a buffer to build the JSON string
    int offset = 0;
    
    offset += snprintf(json_buffer + offset, sizeof(json_buffer) - offset, "{\"temperatures\":[");
    for(int i=0; i < totalDetectedSensors; i++) {
        if (currentTemperatures[i] <= -127.0f) {
            offset += snprintf(json_buffer + offset, sizeof(json_buffer) - offset, "\"N/A\"");
        } else if (currentTemperatures[i] == 85.0f && abs(tempCalibration[i])<0.01f) {
            offset += snprintf(json_buffer + offset, sizeof(json_buffer) - offset, "\"85.0?\"");
        } else {
            offset += snprintf(json_buffer + offset, sizeof(json_buffer) - offset, "\"%.1f\"", currentTemperatures[i]);
        }
        if (i < totalDetectedSensors - 1) offset += snprintf(json_buffer + offset, sizeof(json_buffer) - offset, ",");
    }
    offset += snprintf(json_buffer + offset, sizeof(json_buffer) - offset, "],");
    
    offset += snprintf(json_buffer + offset, sizeof(json_buffer) - offset, 
        "\"uptime\":\"%s\",\"network_status\":\"%s\",\"ntp_time\":\"%s\","
        "\"alarm_active\":%s,\"free_heap_kb\":\"%.1f\"}",
        formatUptime(millis()).c_str(), networkStatusString.c_str(), currentTimeString.c_str(),
        alarmActive ? "true" : "false", (float)ESP.getFreeHeap() / 1024.0f);

    webServer.send(200, "application/json", json_buffer);
}

// Хелперы для парсинга аргументов формы
void getArg(const char* key, char* dest, size_t maxSize) {
    if (webServer.hasArg(key)) {
        strncpy(dest, webServer.arg(key).c_str(), maxSize - 1);
        dest[maxSize - 1] = '\0';
    }
}
void getArg(const char* key, IPAddress& dest) {
    if (webServer.hasArg(key)) dest.fromString(webServer.arg(key));
}
template <typename T>
void getArg(const char* key, T& dest) {
    if (webServer.hasArg(key)) {
        if (std::is_integral<T>::value) dest = webServer.arg(key).toInt();
        else if (std::is_floating_point<T>::value) dest = webServer.arg(key).toFloat();
    }
}

    getArg("graph_thresh_2", newConfig.graphTempThresholds[2]);
    getArg("graph_color_0", newConfig.graphTempColors[0]);
    getArg("graph_color_1", newConfig.graphTempColors[1]);

        getArg(Sprintf("sensor_name_%d", i).c_str(), newConfig.sensorNames[i], sizeof(newConfig.sensorNames[0]));

    if (!needsReboot) applyConfig();

    webServer.send(302, "text/plain", "");
}

void handleCalibratePost() {
    bool changesMade = false;
    for (int i = 0; i < totalDetectedSensors; i++) {
        String key = Sprintf("actual_temp%d", i);
        if(webServer.hasArg(key)) {
            // Recalculate raw temperature from the currently displayed value to avoid race conditions
            float displayed_temp_when_submitted = currentTemperatures[i]; 
            if(displayed_temp_when_submitted > -127.0f && displayed_temp_when_submitted != 85.0f) {
                float raw_temp_measured = displayed_temp_when_submitted - currentConfig.calOffsets[i];
                float desired_temp = webServer.arg(key).toFloat();
                float new_offset = desired_temp - raw_temp_measured;
                if (abs(new_offset) <= 20.0f && abs(currentConfig.calOffsets[i] - new_offset) > 0.001f) { 
                    currentConfig.calOffsets[i] = new_offset;
                    changesMade = true; 
                }
            }
        }
    }
    if (changesMade) { 
        saveConfig(false); 
        applyConfig(); 
        webServerMessage = "Calibration applied and saved."; 
    } else { 
        webServerMessage = "No new calibration changes detected."; 
    }
    webServer.sendHeader("Location", "/settings", true);
    webServer.send(302, "text/plain", "");
}

void handleUpdate() {
  webServer.sendHeader("Connection", "close");
  webServer.send(200, "text/plain", (Update.hasError()) ? "UPDATE FAILED" : "OK");
  delay(1000);
  ESP.restart();
}

void handleUpdateUpload() {
  HTTPUpload& upload = webServer.upload();
  if (upload.status == UPLOAD_FILE_START) {
    Serial.printf("Update Start: %s\n", upload.filename.c_str());
    isOtaUpdating = true;
    tft.fillScreen(ST77XX_BLACK);
    tft.setTextColor(ST77XX_WHITE, ST77XX_BLACK);
    tft.setTextSize(2); tft.setCursor(20, 120); tft.println("WEB OTA UPDATE");
    tft.setTextSize(1); tft.setCursor(20, 150); tft.println("Receiving firmware...");
    if (!Update.begin(UPDATE_SIZE_UNKNOWN, U_FLASH)) Update.printError(Serial);
  } else if (upload.status == UPLOAD_FILE_WRITE) {
    if (Update.write(upload.buf, upload.currentSize) != upload.currentSize) Update.printError(Serial);
    int percentage = (Update.progress() * 100) / (Update.size() > 0 ? Update.size() : 1);
    tft.fillRect(20, 165, 200, 12, ST77XX_DARKGREY);
    tft.fillRect(20, 165, constrain(percentage * 2, 0, 200), 12, ST77XX_GREEN);
    tft.drawRect(20, 165, 200, 12, ST77XX_WHITE);
  } else if (upload.status == UPLOAD_FILE_END) {
    if (Update.end(true)) {
      Serial.printf("Update Success: %u bytes\n", upload.totalSize);
      tft.setCursor(20, 180); tft.println("Update Finished! Rebooting...");
    } else {
      Update.printError(Serial);
      tft.fillScreen(ST77XX_RED);
      tft.setTextColor(ST77XX_WHITE);
      tft.setTextSize(2); tft.setCursor(20, 140); tft.println("Update FAILED!");
    }
    delay(2000);
    isOtaUpdating = false;
  }
}

void handleNotFound() {
    if(apModeIsActive) {
        webServer.sendHeader("Location", String("http://") + WiFi.softAPIP().toString(), true);
        webServer.send(302, "text/plain", "");
        return;
    }
    String message = "404 Not Found\nURI: " + webServer.uri();
    webServer.send(404, "text/plain", message);
}

    page += Sprintf("<tr><td>Thresh 2 ≤ Temp < Alarm</td><td data-label='Threshold'><input type='number' step='0.1' name='graph_thresh_2' value='%.1f'></td><td>%s</td></tr>", currentConfig.graphTempThresholds[2], serveColorSelect("graph_color_2", currentConfig.graphTempColors[2]).c_str());
    page += Sprintf("<tr><td>Temp ≥ Alarm</td><td>-</td><td>%s</td></tr>", serveColorSelect("graph_color_3", currentConfig.graphTempColors[3]).c_str());
    page += F("</tbody></table>");
    
    page += F("<p style='text-align:center;'><input type='submit' value='Save All Settings' class='submit-button'></p><p class='info-text'>Device will reboot if Device Name or network configuration changes.</p></form></div>");
    return page;
}

String serveCalibrationSection() {
    String page;
    page += F("<div class='form-section'><h2>Temperature Sensors & Calibration</h2>");
    if (totalDetectedSensors > 0) {
        page += F("<p class='info-text'>Allow sensor to stabilize, enter the real temperature, and submit.</p>");
        page += F("<form method='POST' action='/calibrate'><table id='calibration_table'><thead><tr><th>Sensor</th><th>Raw</th><th>Offset</th><th>Displayed</th><th>Enter Actual</th></tr></thead><tbody>");
        for (int i = 0; i < totalDetectedSensors; i++) {
            float raw_reading = currentTemperatures[i] - currentConfig.calOffsets[i];
            bool isErr_current = (raw_reading <= -127.0f || raw_reading == 85.0f);
            
            page += Sprintf("<tr><td data-label='Sensor'>S%d %s</td>", i+1, (strlen(currentConfig.sensorNames[i]) > 0 ? ("("+String(currentConfig.sensorNames[i])+")").c_str() : ""));
            page += Sprintf("<td data-label='Raw'>%s</td>", isErr_current ? "Error" : String(raw_reading, 1).c_str());
            page += Sprintf("<td data-label='Offset'>%.2f</td>", currentConfig.calOffsets[i]);
            page += Sprintf("<td data-label='Displayed'>%s</td>", isErr_current ? "Error" : String(currentTemperatures[i], 1).c_str());
            page += Sprintf("<td data-label='Enter Actual'><input type='text' class='cal' name='actual_temp%d' value='%s'></td></tr>", i, isErr_current ? "" : String(currentTemperatures[i],1).c_str());
        }
        page += F("</tbody></table><p style='text-align:center;'><input type='submit' value='Apply & Save Calibration' class='submit-button'></p></form>");
    } else {
        page += F("<p style='text-align:center;'>No sensors detected.</p>");
    }
    page += F("</div>");
    return page;
}
